```bash
# No shebang strictness; composes under `source` cleanly.
log() {
  local message="${1:-}"
  if [[ -n "${LOG_FILE:-}" ]]; then

  else

  fi
}

die() {
  local message="${1:-Fatal error}"
  log "ERROR: $message"
  exit 1
}
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| file-system -> library | YAML file contents enter the library via yq subshell; malformed YAML must fail closed |
| caller-args -> library | Function args include file paths and jq program strings from calling scripts |
| yq subshell -> bash | yq output (JSON) is piped into jq and potentially into shell expansions |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01-01 | Tampering | `read_yaml_file` malformed YAML | mitigate | yq exits non-zero on invalid YAML; wrap in `yq -o=json "$path" || die "Failed to parse YAML: $path"` so scorer never sees partial output |
| T-02-01-02 | Denial of Service | `merge_yaml_defaults` with deeply-nested or cyclic YAML | accept | yq handles recursion internally; case YAMLs are repo-controlled in v1.2; PEL-era hardening lives in Phase 7 code proposer's sandbox |
| T-02-01-03 | Information Disclosure | `atomic_json_write` temp file leakage | mitigate | use `mktemp` (secure random name, default 0600 perms); `trap` not required inside the library (callers set their own traps per PATTERNS.md) |
| T-02-01-04 | Tampering | `render_report` placeholder injection via `{{KEY}}` values containing `{{`-sequences | mitigate | substitute one key at a time with pure parameter expansion `${rendered//\\{\\{KEY\\}\\}/$value}`; do NOT re-scan the rendered output for more placeholders |
| T-02-01-05 | Elevation of Privilege | Shell-eval of YAML content | mitigate | the library NEVER passes YAML content to eval / source / bash -c; yq-produced JSON is consumed only via jq (which does not exec) and through `--arg` / `--argjson` when interpolating |
| T-02-01-06 | Repudiation | Silent failure of jq in atomic_json_write | mitigate | atomic_json_write MUST die with explicit jq-stderr capture: `jq "$prog" "$in" > "$tmp" 2>"$tmp.err" || { err=$(<"$tmp.err"); rm -f "$tmp" "$tmp.err"; die "jq failed in atomic_json_write: $err"; }` |
</threat_model>

       ```bash
       render_report() {
         local template_path="${1:?render_report requires a template path}"
         shift
         [[ -f "$template_path" ]] || die "Template file not found: $template_path"
         local rendered
         rendered=$(<"$template_path")
         local pair key value
         for pair in "$@"; do
           key="${pair%%=*}"
           value="${pair#*=}"
           # Parameter expansion: replace every literal {{KEY}} with value.
           # Using ${var//pattern/replacement} — braces must be escaped with backslash for literal match.
           rendered="${rendered//\{\{$key\}\}/$value}"
         done
         printf '%s' "$rendered"
       }
       ```

    6. Atomic JSON writes (per PATTERNS.md §Shared Patterns "jq Atomic Writes" — drop command -v guard per D-03 since jq is mandatory):
       ```bash
       atomic_json_write() {
         local jq_program="${1:?atomic_json_write requires a jq program}"
         local input_path="${2:?atomic_json_write requires an input path}"
         local output_path="${3:?atomic_json_write requires an output path}"
         ensure_jq
         [[ -f "$input_path" ]] || die "atomic_json_write: input not found: $input_path"
         local tmp err
         tmp=$(mktemp)
         err=$(jq "$jq_program" "$input_path" > "$tmp" 2>&1) && {
           mv "$tmp" "$output_path"
           return 0
         }
         rm -f "$tmp"
         die "atomic_json_write: jq failed: $err"
       }

       atomic_json_write_stdin() {
         local jq_program="${1:?atomic_json_write_stdin requires a jq program}"
         local output_path="${2:?atomic_json_write_stdin requires an output path}"
         ensure_jq
         local tmp err
         tmp=$(mktemp)
         err=$(jq "$jq_program" > "$tmp" 2>&1) && {
           mv "$tmp" "$output_path"
           return 0
         }
         rm -f "$tmp"
         die "atomic_json_write_stdin: jq failed: $err"
       }
       ```

    5. **Dimension: Cost** (port score-run.ps1:380-402):
       - Wall clock = last phase end_time - first phase start_time from `state.history[]` (or fall back to state.started_at / state.completed_at if history missing).
       - Output bytes = sum of sizes of `$outputs_dir/*.log` and `$outputs_dir/*.txt` (or equivalent globs per PS).
       - Compare wall_clock vs `case.expectations.cost.max_wall_clock_seconds` (default 900 from defaults.yaml).
       - Compare output_bytes vs `case.expectations.cost.max_total_output_bytes` if set; else skip the bytes check.
       - Result: PASS if both under threshold; PARTIAL if one over, one under; FAIL if both over OR wall_clock is missing/unparseable.

       Concrete jq extraction:
       ```bash
       wall_clock_max=$(jq -r '.expectations.cost.max_wall_clock_seconds // 900' <<<"$case_json")
       output_bytes_max=$(jq -r '.expectations.cost.max_total_output_bytes // "unset"' <<<"$case_json")

       # Wall-clock: prefer state.history's first start + last end; fall back to state.started_at / state.completed_at.
       started_at=$(jq -r '(.history[0].started_at // .started_at // "")' "$state_json_path")
       ended_at=$(jq -r '((.history[-1].ended_at // .history[-1].completed_at) // .completed_at // "")' "$state_json_path")
       wall_secs=""
       if [[ -n "$started_at" && -n "$ended_at" ]]; then
         # Use jq for ISO8601 → epoch diff (portable across GNU/BSD date).
         wall_secs=$(jq -rn --arg s "$started_at" --arg e "$ended_at" '
           ($s | fromdateiso8601? // empty) as $s_epoch |
           ($e | fromdateiso8601? // empty) as $e_epoch |
           if $s_epoch and $e_epoch then ($e_epoch - $s_epoch) else "" end
         ')
       fi

       # Output bytes
       total_bytes=0
       if [[ -d "$outputs_dir" ]]; then
         while IFS= read -r f; do
           [[ -f "$f" ]] && total_bytes=$((total_bytes + $(wc -c < "$f")))
         done < <(find "$outputs_dir" -maxdepth 1 -type f \( -name '*.log' -o -name '*.txt' \))
       fi

       # Scoring
       cost="PASS"
       wc_over=false
       if [[ -z "$wall_secs" ]]; then
         wc_over=true   # Missing timing → treat as over-budget for determinism
       elif (( wall_secs > wall_clock_max )); then
         wc_over=true
       fi
       bytes_over=false
       if [[ "$output_bytes_max" != "unset" && "$output_bytes_max" != "null" ]]; then
         (( total_bytes > output_bytes_max )) && bytes_over=true
       fi
       if [[ "$wc_over" == "true" && "$bytes_over" == "true" ]]; then
         cost="FAIL"
       elif [[ "$wc_over" == "true" || "$bytes_over" == "true" ]]; then
         cost="PARTIAL"
       fi
       ```

       scores_json=$(jq -n \

         {cross_ai_diversity:1, convergence:1, plan_quality:1, execution_fidelity:1, verify_accuracy:1, cost:1, robustness:2} as $W |
         {PASS:1.0, PARTIAL:0.5, FAIL:0.0, "N/A":null} as $V |
         ($scores | to_entries | map(
           ($W[.key]) as $w | ($V[.value]) as $v |
           if $v == null then null else {sum_w:$w, sum_s: ($w * $v)} end
         ) | map(select(.!=null))) as $rows |

       # Each input file contains a JSON array of strings. Emits a float in [0.0, 1.0] on stdout.

       ```bash
       # levenshtein <a_file> <b_file>
       # Emits integer edit distance on stdout. 4000-char cap per PS (prevents OOM + slow DP).
       levenshtein() {
         local a_file="${1:?levenshtein requires path a}"
         local b_file="${2:?levenshtein requires path b}"
         awk -v a_file="$a_file" -v b_file="$b_file" '
           BEGIN {
             # Read both files; truncate to 4000 chars (PS cap).
             getline a < a_file;  while ((getline l < a_file) > 0) a = a "\n" l
             getline b < b_file;  while ((getline l < b_file) > 0) b = b "\n" l
             close(a_file); close(b_file)
             la = length(a); if (la > 4000) { la = 4000; a = substr(a, 1, 4000) }
             lb = length(b); if (lb > 4000) { lb = 4000; b = substr(b, 1, 4000) }
             if (la == 0) { print lb; exit }
             if (lb == 0) { print la; exit }
             # Standard Wagner–Fischer 2-row optimization.
             for (j = 0; j <= lb; j++) prev[j] = j
             for (i = 1; i <= la; i++) {
               curr[0] = i
               ca = substr(a, i, 1)
               for (j = 1; j <= lb; j++) {
                 cb = substr(b, j, 1)
                 cost = (ca == cb) ? 0 : 1
                 del_ = prev[j]   + 1
                 ins  = curr[j-1] + 1
                 sub_ = prev[j-1] + cost
                 m = del_; if (ins < m) m = ins; if (sub_ < m) m = sub_
                 curr[j] = m
               }
               for (j = 0; j <= lb; j++) prev[j] = curr[j]
             }
             print prev[lb]
           }
         '
       }
       ```
